def fixed_spent50_predecessor_fixture():
    """Read the fixed private provider projection; bodies never enter test output."""
    import os, stat, json, hashlib
    from pathlib import Path
    import v03_dogfood_full_composition as c
    cached = globals().get("_SPENT50_PREDECESSOR_FIXTURE_CACHE")
    if cached is not None:
        return dict(cached)
    directory = Path(os.environ["V03_SPENT50_PREDECESSOR_FIXTURE_DIR"])
    expect(directory.is_absolute(), "spent50 fixture path is not absolute")
    fd = os.open(directory, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        before = os.fstat(fd)
        expect(stat.S_IMODE(before.st_mode) == 0o700 and before.st_uid == os.getuid(),
               "spent50 fixture directory is not private")
        names = {"run.json", "jobs.json", "artifacts.json", "issue.json",
                 "agent.txt", "detection.txt", "publication.txt", "upload.txt"}
        entries = []
        with os.scandir(fd) as iterator:
            for entry in iterator:
                entries.append(entry.name)
                expect(len(entries) <= 9, "spent50 fixture directory exceeds bound")
        expect(set(entries) == names | {"manifest.json"}, "spent50 fixture files differ")
        def read(name, limit):
            item = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fd)
            try:
                info = os.fstat(item)
                expect(stat.S_ISREG(info.st_mode) and info.st_nlink == 1
                       and info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o600
                       and 0 < info.st_size <= limit, "spent50 fixture file shape differs")
                with os.fdopen(item, "rb", closefd=False) as stream:
                    raw = stream.read(limit + 1)
                after = os.fstat(item)
                expect(len(raw) == info.st_size and len(raw) <= limit
                       and (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns) ==
                           (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns),
                       "spent50 fixture changed during bounded read")
                return raw
            finally:
                os.close(item)
        def strict(raw):
            def pairs(rows):
                result = {}
                for key, value in rows:
                    expect(key not in result, "spent50 fixture JSON duplicate key")
                    result[key] = value
                return result
            return json.loads(raw, object_pairs_hook=pairs)
        manifest = strict(read("manifest.json", 16384))
        expect(set(manifest) == {"schema_version", "run_id", "source_sha", "files"}
               and manifest["schema_version"] == "ai-sdlc.spent50-predecessor-fixture/v1"
               and type(manifest["run_id"]) is int and manifest["run_id"] == 38699515120
               and manifest["source_sha"] == c.SPENT50_FAILURE_RUN["head_sha"]
               and set(manifest["files"]) == names, "spent50 fixture manifest differs")
        result = {}
        for name in sorted(names):
            row = manifest["files"][name]
            expect(set(row) == {"size", "sha256"} and type(row["size"]) is int,
                   "spent50 fixture manifest entry differs")
            raw = read(name, 256 * 1024)
            expect(len(raw) == row["size"] and hashlib.sha256(raw).hexdigest() == row["sha256"],
                   "spent50 fixture component digest differs")
            if name.endswith(".json"):
                strict(raw)
            result[name] = raw
        issue = strict(result["issue.json"])
        expect(issue["id"] == 5862328383 and issue["number"] == 606 and type(issue["body"]) is str
               and hashlib.sha256(issue["body"].encode()).hexdigest() ==
                   "2cd34237f302cd6174ba16fac4431b8331c651e6186608552221675bdab7749f",
               "spent50 fixture issue is not the original pinned bytes")
        for name, expected in (("agent.txt", c.SPENT50_FAILURE_AGENT_LINES),
                               ("detection.txt", c.SPENT50_FAILURE_DETECTION_LINES),
                               ("publication.txt", c.SPENT50_FAILURE_PUBLICATION_LINES),
                               ("upload.txt", c.SPENT50_FAILURE_UPLOAD_LINES)):
            lines = result[name].decode("utf-8").splitlines()
            expect(lines == list(expected), "spent50 fixture selected original lines differ")
        after = directory.lstat()
        expect(not stat.S_ISLNK(after.st_mode)
               and (before.st_dev, before.st_ino) == (after.st_dev, after.st_ino),
               "spent50 fixture directory identity changed")
        for name in names | {"manifest.json"}:
            os.unlink(name, dir_fd=fd)
        globals()["_SPENT50_PREDECESSOR_FIXTURE_CACHE"] = result
        return dict(result)
    finally:
        os.close(fd)
